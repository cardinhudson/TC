# ==========================================
# IMPORTANTE: VERSÕES ESPECÍFICAS OBRIGATÓRIAS
# ==========================================
# Este arquivo usa versões EXATAS (==) em vez de >= para garantir
# que todos os PCs tenham exatamente as mesmas versões instaladas.
#
# PROBLEMA COMUM: Se usar >=, diferentes PCs podem instalar versões
# diferentes das bibliotecas, causando problemas de renderização de
# tabelas, gráficos e comportamento inconsistente do Streamlit.
#
# SOLUÇÃO: Sempre instale as versões EXATAS especificadas abaixo.
# Se houver problemas, desinstale tudo e reinstale usando este arquivo:
#   pip uninstall streamlit pandas numpy altair plotly openpyxl -y
#   pip install -r requirements.txt
# ==========================================

streamlit==1.50.0
# Versão específica necessária para compatibilidade com st.expander
# Versões anteriores podem causar erros ao usar expanders
# NOTA: O código foi ajustado para evitar expanders aninhados em 3 níveis
# (substituídos por containers no nível 3) para evitar problemas de renderização

pandas==2.3.3
# Versão específica para garantir renderização consistente de tabelas
# Versões diferentes podem causar problemas no st.dataframe() e na
# formatação de DataFrames, especialmente com tipos category e downcast

numpy==2.3.5
# Versão específica compatível com pandas 2.3.3
# Mudanças no numpy podem afetar processamento de dados numéricos

altair==5.5.0
# Versão específica para gráficos Altair
# Compatível com Streamlit 1.50.0

plotly==6.5.0
# Versão específica para gráficos Plotly
# Versões diferentes podem causar problemas de renderização

openpyxl==3.1.5
# Versão específica para leitura/escrita de arquivos Excel
# Necessário para processar arquivos .xlsx

# ==========================================
# DEPENDÊNCIAS DO CHATBOT (Opcionais - Fallback disponível)
# ==========================================
# Estas bibliotecas são opcionais. Se não instaladas, o chatbot usará
# método de busca simples baseado em palavras-chave.
# Para melhor qualidade de respostas, recomenda-se instalar todas.

sentence-transformers==2.2.2
# Biblioteca para criar embeddings vetoriais de texto
# Usado no chatbot para busca semântica local (sem APIs externas)
# Modelo multilingual suporta português
# Primeira execução baixa modelo ~420MB (depois usa cache local)

faiss-cpu==1.7.4
# Biblioteca Facebook AI Similarity Search para busca rápida em vetores
# Versão CPU (sem GPU) - adequada para uso local
# Usado para busca eficiente de similaridade semântica no chatbot
# NOTA: Se houver problemas de instalação, tente: pip install faiss-cpu --no-cache-dir

torch==2.0.1
# PyTorch - dependência do sentence-transformers
# Versão compatível com sentence-transformers 2.2.2
# Tamanho: ~500MB (dependência pesada mas necessária)

transformers==4.35.2
# Hugging Face Transformers - dependência do sentence-transformers
# Versão compatível com sentence-transformers 2.2.2
